# Add Dex to Your Agent

Move the example agent of the [Agent Quick Start](/agent-frameworks/quick-start) into your own code: install your framework's adapter, copy the complete files, and serve the session from a web app.

<CodingAgentBox prompt="Follow https://docs.superdurable.io/agent-frameworks/add-to-your-agent.md to make the agent in this repository durable with Dex." />

## What changes in your app {#what-changes-in-your-app}

- **Your agent definition stays.** You wrap it and give each tool a policy.
- **Your request handler changes shape.** It sends the user's message to the session and returns. The reply arrives on the session's event stream, or in its snapshot.
- **You run two more things.** A Worker process runs the session Steps, and Dex Server stores them. Dex Server runs in Dex Cloud or on your own infrastructure; [Where to run Dex Server](/components#where-to-run) compares the two. Dex is in Beta, Dex Server is at version 0.1, and one active node serves a cluster.
- **Each turn waits on Dex commits.** A turn with three parallel tool calls adds about 0.4 seconds; see [Latency and cost](/agent-frameworks/sessions#latency-and-cost).

[Run Agents in Production](/agent-frameworks/production) lists what to set up. [Performance](/intro/performance) gives the numbers, and [Limits](/intro/limits#license) gives the license.

### Wire the session into your app {#wire-it-into-your-app}

The change to your agent is small: the wrap in [step 2 of the Agent Quick Start](#what-changes-in-your-code). The rest is hosting code that every Dex app has, a Worker that runs Steps and a Client that calls them. This function is the smallest host. It runs one Worker and one Client in your process, starts a session, sends one message, and returns the session after the turn:

<SdkTabs
  examples={{
    python: 'examples/python/agent-harnesses/agent_harness_examples/quick_start/embed.py',
    typescript: 'examples/typescript/src/products/agent-harnesses/quick-start/quick-start-embed.ts',
  }}>
<SdkSnippet lang="python">

```python
"""The smallest host for a durable agent: one Worker, one Client, one turn."""

from __future__ import annotations

import asyncio

from dex import (
    AsyncClient,
    AsyncWorker,
    BlobCacheConfig,
    ClientOptions,
    Registry,
    WorkerOptions,
    open_blob_cache,
)
from dex_agents import AgentSessionClient, AgentSessionFlow, AgentSessionStatus, AgentSnapshot

async def ask_agent(
    session_flow: AgentSessionFlow,
    session_id: str,
    message: str,
    server_address: str = "127.0.0.1:8801",
    blob_cache_dir: str = ".dex-blob-cache",
) -> AgentSnapshot:
    """Starts a session, sends one message, and returns the session after the turn."""
    registry = Registry((session_flow,), allow_async_handlers=True)
    cache = open_blob_cache(BlobCacheConfig(blob_cache_dir, 1 << 30))
    worker = AsyncWorker(registry, cache, WorkerOptions(server_address=server_address))
    client = AsyncClient(registry, cache, ClientOptions(server_address=server_address))
    worker_task = asyncio.create_task(worker.start())
    try:
        await worker.wait_pull_registered(timeout=30)
        sessions = AgentSessionClient(client, session_flow)
        await sessions.start_session(session_id)
        await sessions.send_user_message(session_id, message)
        while True:
            snapshot = await sessions.get_agent_snapshot(session_id)
            state = snapshot.state
            if state and state.turn_count >= 1:
                if state.status == AgentSessionStatus.WAITING_FOR_USER_MESSAGE:
                    return snapshot
            if state and state.status == AgentSessionStatus.FAILED:
                raise RuntimeError(f"the session failed: {state.last_turn_error}")
            await asyncio.sleep(0.5)
    finally:
        await client.close()
        await worker.close()
        await worker_task
        cache.close()
```

```ts
// The smallest host for a durable agent: one Worker, one Client, one turn.

import { Client, Registry, Worker, openBlobCache } from "@superdurable/dex";
import {
  readAgentSnapshot,
  sendAgentUserMessage,
  startAgentSession,
  type AgentSessionFlow,
  type AgentSnapshot,
} from "@superdurable/dex-agents";

/** Starts a session, sends one message, and returns the session after the turn. */
export async function askAgent(
  flow: AgentSessionFlow,
  sessionId: string,
  message: string,
  serverAddress = "127.0.0.1:8801",
  blobCacheDir = ".dex-blob-cache",
): Promise<AgentSnapshot> {
  const registry = new Registry([flow]);
  const blobCache = openBlobCache({ directory: blobCacheDir, maxBytes: 1 << 30 });
  const worker = new Worker(registry, blobCache, { serverAddress });
  const client = new Client(registry, blobCache, { serverAddress });
  try {
    await worker.start();
    await worker.waitPullRegistered({ timeoutMs: 30_000 });
    await startAgentSession(client, flow, sessionId, {});
    await sendAgentUserMessage(client, flow, sessionId, { text: message });
    for (;;) {
      const snapshot = await readAgentSnapshot(client, flow, sessionId);
      const state = snapshot.state;
      if (state?.status === "failed") {
        throw new Error(`the session failed: ${state.lastTurnError ?? "unknown error"}`);
      }
      if (state !== null && state.turnCount >= 1 && state.status === "waiting_for_user_message") {
        return snapshot;
      }
      await new Promise((resolve) => setTimeout(resolve, 500));
    }
  } finally {
    await client.close();
    await worker.close();
    blobCache.close();
  }
}
```

It takes the session Flow that your module's `build_durable_agent` or `buildDurableAgent` returns. A turn that needs an approval waits until your app resolves it; [Approve from your application](/agent-frameworks/approvals#approve-from-your-application) shows how.

A web app splits the same pieces across routes:

1. A **Registry** with your session Flow, a **Worker** that runs its Steps, and a **Client** that calls it, as in the function above. The Worker connects out to Dex, so it needs no open port. Both open a blob cache, a local cache for large values; [Registry and blob cache](/components/application#blob-cache) explains it.
2. A route that starts a session. The session ID is the Flow ID.
3. A route that sends a user message, and one that reads the session snapshot with the agent's reply. To stream the reply as it is written, read the AgentEvents Stream, as [the next section](#web-handler) shows.
4. A route that approves or rejects a waiting tool call. [Approve from your application](/agent-frameworks/approvals#approve-from-your-application) covers it.

The [Order Support Agent](/use-cases/order-support-agent) example has these routes over HTTP, for every framework.

```python
"""A complete web app for a durable agent: start, message, read, stream, approve.

It hosts the Worker and the Client in the web process. Add your own
authentication before you expose these routes: anyone who knows a session ID
can read and change that session here.
"""

from __future__ import annotations

import uuid
from pathlib import Path
from typing import Any

from dex_agents import AgentSessionFlow
from dex_agents.ui_stream import (
    AGENT_EVENT_STREAM_HEADERS,
    agent_snapshot_to_json,
    stream_agent_event_frames,
)
from quart import Quart, Response, jsonify, make_response, request

from agent_harness_examples.quick_start.safe_send import send_user_message_once
from agent_harness_examples.quick_start.session import QuickStartApp

def create_web_app(
    session_flow: AgentSessionFlow, server_address: str, blob_cache_dir: Path
) -> Quart:
    """Returns a Quart app whose routes run one durable agent's sessions."""
    web = Quart(__name__)
    apps: list[QuickStartApp] = []

    def dex() -> QuickStartApp:
        return apps[0]

    # The Worker and the Client bind to the event loop that serves the app.
    @web.before_serving
    async def start_worker() -> None:
        apps.append(QuickStartApp(session_flow, server_address, blob_cache_dir))
        await dex().start_worker()

    @web.after_serving
    async def stop_worker() -> None:
        await apps.pop().close()

    @web.post("/sessions")
    async def start_session() -> Response:
        session_id = f"web-{uuid.uuid4().hex}"
        await dex().sessions.start_session(session_id)
        return jsonify(session_id=session_id)

    @web.post("/sessions/<session_id>/messages")
    async def send_message(session_id: str) -> Response:
        body: dict[str, Any] = await request.get_json()
        await send_user_message_once(dex().sessions, session_id, str(body["text"]))
        return jsonify(accepted=True)

    @web.get("/sessions/<session_id>")
    async def read_session(session_id: str) -> Response:
        return jsonify(agent_snapshot_to_json(await dex().sessions.get_agent_snapshot(session_id)))

    @web.get("/sessions/<session_id>/events")
    async def stream_events(session_id: str) -> Response:
        resume_token = request.headers.get("Last-Event-ID", "")
        frames = await stream_agent_event_frames(
            dex().client, session_flow, session_id, resume_token=resume_token
        )
        response = await make_response(frames, dict(AGENT_EVENT_STREAM_HEADERS))
        assert isinstance(response, Response)
        response.timeout = None
        return response

    @web.post("/sessions/<session_id>/interrupts/<interrupt_id>")
    async def resolve(session_id: str, interrupt_id: str) -> Response:
        body: dict[str, Any] = await request.get_json()
        accepted = await dex().sessions.resolve_interrupt(
            session_id, interrupt_id, approved=body.get("approved") is True
        )
        return jsonify(accepted=accepted)

    return web
```
