# Polling with Timer

**PollingWithTimerFlow** makes the wait explicit. **PollingStep** waits for a Timer, checks the external condition, then either completes or moves to itself for another interval. It uses more code than retry-based polling, but the not-ready outcome is a normal Step decision rather than a failed attempt.

```go
type pollingWithTimerStep struct {
	dex.StepDefaults
}

func (pollingWithTimerStep) GetStepType() string { return "PollingStep" }

func (pollingWithTimerStep) WaitFor(
	ctx dex.Context,
	input dex.None,
) (*dex.Wait, error) {
	return dex.Until(dex.Timer(10 * time.Second)), nil
}

func (pollingWithTimerStep) Execute(
	ctx dex.Context,
	input dex.None,
) (*dex.StepDecision, error) {
	if isSystemReady() {
		return dex.GracefulComplete(nil), nil
	}
	return dex.GoTo(pollingWithTimerStep{}, nil), nil
}
```

```rust
impl Step for PollingStep {
    type Input = u32;
    fn step_type(&self) -> &'static str {
        "PollingStep"
    }

    fn wait_for(&self, _: &mut Context, _: u32) -> HandlerResult<Wait> {
        Ok(Wait::until(Timer::by_duration(Duration::from_secs(5))))
    }

    fn execute(&self, _: &mut Context, remaining: u32) -> HandlerResult<StepDecision> {
        if remaining <= 1 {
            Ok(StepDecision::graceful_complete("ready".to_owned()))
        } else {
            Ok(StepDecision::go_to(&PollingStep, remaining - 1))
        }
    }
}
```
