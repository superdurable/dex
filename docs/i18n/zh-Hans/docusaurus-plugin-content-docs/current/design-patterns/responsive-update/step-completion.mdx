# 等待 Step 完成

当 Flow 在后台工作结束前，会先到达一个客户端已经可以使用的 durable 节点时，使用 `WaitForStepCompletion`。调用方启动 Flow、等待该 Step，然后读取需要的结果。调用方拿到响应后，后续 Step 继续运行。

`WaitForStepCompletionFlow` 示例会在 `PersistData` 中持久化记录，再启动 `BackgroundWork`。它的 controller 在查询已持久化记录前等待 `PersistData`。要等待的 Step 必须在保存好响应所需的状态之后才完成。

如果在 HTTP endpoint 或其他有自己截止时间的入口里执行等待，请设置 `RequestTimeout`。下面的 runnable controller 没有设置它。见[等待超时与上限](/reference/server/runtime-client#wait-timeouts-and-limits)。

func (controller *controller) start(request *gin.Context) {
	flowID, found := httputil.RequiredQuery(request, "workflowId")
	if !found {
		return
	}
	input := JobSeekerData{ID: 1}
	_, err := controller.client.StartFlow(
		request.Request.Context(), controller.flow, flowID, input, patternStartOptions(),
	)
	if err != nil {
		httputil.RespondString(request, "", err)
		return
	}
	waitContext, cancelWait := context.WithTimeout(request.Request.Context(), 5*time.Minute)
	defer cancelWait()
	err = controller.client.WaitForStepCompletion(
		waitContext, flowID, sdk.StepExecutionID{StepType: "PersistData"}, sdk.WaitForStepCompletionOptions{},
	)
	if err != nil {
		httputil.RespondString(request, "", err)
		return
	}
	var persistedData JobSeekerData
	err = controller.client.InvokeRPC(
		request.Request.Context(), flowID, controller.flow.GetJobSeekerData,
		nil, &persistedData,
	)
	if err != nil {
		httputil.RespondString(request, "", err)
		return
	}
	serialized, err := json.Marshal(persistedData)
	if err != nil {
		httputil.RespondString(request, "", err)
		return
	}
	httputil.RespondString(
		request, fmt.Sprintf("success for workflow %s with data %s", flowID, serialized), nil,
	)
}

async fn start(
    State(client): State<SharedClient>,
    Query(query): Query<StartQuery>,
) -> impl IntoResponse {
    let flow_id = if query.workflow_id.is_empty() {
        new_flow_id("wait-state")
    } else {
        query.workflow_id
    };
    match run_blocking(move || {
        let flow = WaitForStepCompletionFlow::default();
        let input = PersistRequest {
            record_id: "1".into(),
            payload: "Test Resume".into(),
        };
        let run_id = client.start_flow(&flow, &flow_id, input)?;
        client.wait_for_step_completion(
            &flow_id,
            WaitForStepCompletionFlow::persisted_step(),
            WaitForStepCompletionOptions::new(),
        )?;
        Ok(StartResponse { flow_id, run_id })
    }) {
        Ok(value) => ok_json(value),
        Err(error) => map_sdk_error(error).into_response(),
    }
}
